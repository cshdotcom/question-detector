    55→export function ExportMenu({ currentImageOnly = false }: ExportMenuProps) {
    56→  const images = useDetectorStore((s) => s.images)
    57→  const currentImageId = useDetectorStore((s) => s.currentImageId)
    58→  const [busy, setBusy] = useState(false)
    59→  const [selectImagesOpen, setSelectImagesOpen] = useState(false)
    60→  const [selectedImageIds, setSelectedImageIds] = useState<Set<string>>(new Set())
    61→  const [pendingAction, setPendingAction] = useState<"pdf" | "word" | "excel" | "markdown" | "json" | "boxes" | "cropped" | null>(null)
    62→  // 导出选项：是否含题目原图截图 / 配图 / 批改结果
    63→  const [optIncludeImage, setOptIncludeImage] = useState(true)
    64→  const [optIncludeSubImages, setOptIncludeSubImages] = useState(true)
    65→  const [optIncludeGrading, setOptIncludeGrading] = useState(true)
    66→
    67→  // 当前要导出的图片列表
    68→  const targetImages = currentImageOnly
    69→    ? images.filter((im) => im.id === currentImageId)
    70→    : selectedImageIds.size > 0
    71→    ? images.filter((im) => selectedImageIds.has(im.id))
    72→    : images
    73→
    74→  // 打开"选择图片+选项"对话框
    75→  const openSelectImages = (action: "pdf" | "word" | "excel" | "markdown" | "json" | "boxes" | "cropped") => {
    76→    if (images.length === 0) {
    77→      toast.error("没有图片可导出")
    78→      return
    79→    }
    80→    setPendingAction(action)
    81→    setSelectedImageIds(new Set(images.map((im) => im.id)))
    82→    setSelectImagesOpen(true)
    83→  }
    84→
    85→  const confirmSelectAndExport = async () => {
    86→    setSelectImagesOpen(false)
    87→    if (selectedImageIds.size === 0) {
    88→      toast.error("请至少选择一张图片")
    89→      return
    90→    }
    91→    const targets = images.filter((im) => selectedImageIds.has(im.id))
    92→    if (pendingAction) {
    93→      await runExport(pendingAction, targets)
    94→    }
    95→    setPendingAction(null)
    96→  }
    97→
    98→  const toggleImage = (id: string) => {
    99→    setSelectedImageIds((prev) => {
   100→      const next = new Set(prev)
   101→      if (next.has(id)) next.delete(id)
   102→      else next.add(id)
   103→      return next

   105→  }

   107→  const selectAll = () => setSelectedImageIds(new Set(images.map((im) => im.id)))

   110→  const runExport = async (
   111→    action: "pdf" | "word" | "excel" | "markdown" | "json" | "boxes" | "cropped",
   112→    targets: typeof images
   113→  ) => {
   114→    if (targets.length === 0) {
   115→      toast.error("没有图片可导出")
   116→      return
   117→    }
   118→    // 选项仅对 pdf/word/cropped 有意义；markdown/excel/json 始终含批改文本；boxes 始终含图
   119→    if (action === "json") return exportJson(targets)
   120→    if (action === "markdown") return exportMarkdown(targets)
   121→    if (action === "excel") return exportExcel(targets)
   122→    if (action === "boxes") return exportWithBoxes(targets, true)
   123→    if (action === "cropped") return exportCroppedImages(targets, optIncludeSubImages)
   124→    if (action === "pdf") return exportPDF(targets)
   125→    if (action === "word") return exportWord(targets)
   126→  }

   128→  // ===== JSON =====
   129→  const exportJson = (targets: typeof images) => {

   137→  // ===== Markdown =====
   138→  const exportMarkdown = (targets: typeof images) => {
   139→    const lines: string[] = ["# 题目识别结果", ""]
   140→    targets.forEach((img, i) => {
   141→      lines.push(`## 图片 ${i + 1}：${img.name}`)
   142→      lines.push("")
   143→      const passages = img.questions.filter((q) => q.kind === "passage")
   144→      passages.forEach((p) => {
   145→        lines.push(`### 阅读材料`)
   146→        lines.push("")
   147→        lines.push(p.content || "（无内容）")
   148→        lines.push("")
   149→      })
   150→      if (img.questions.length === 0) {
   151→        lines.push("_未识别到题目_")
   152→      } else {
   153→        img.questions
   154→          .filter((q) => q.kind !== "passage")

   156→            const num = q.number ? `（${q.number}）` : `（${qi + 1}）`
   157→            lines.push(`### 第 ${qi + 1} 题 ${num}`)
   158→            lines.push("")
   159→            if (q.kind === "passage_question" && q.passageId) {
   160→              const passage = img.questions.find((x) => x.id === q.passageId)
   161→              if (passage) {
   162→                lines.push("> **阅读材料：**")
   163→                lines.push(">")
   164→                lines.push(`> ${passage.content.replace(/\n/g, "\n> ")}`)
   165→                lines.push("")
   166→              }
   167→            }
   168→            lines.push(q.content || "_（无内容）_")
   169→            lines.push("")
   170→            if (q.images.length > 0) {
   171→              lines.push(`> 配图 ${q.images.length} 张`)
   172→              lines.push("")
   173→            }
   174→            if (q.grading) {
   175→              lines.push(`**批改结果：** ${q.grading.correct === true ? "✓ 正确" : q.grading.correct === "partial" ? "△ 部分正确" : "✗ 错误"}`)
   176→              if (q.grading.correctAnswer) {
   177→                lines.push(`**正确答案：** ${q.grading.correctAnswer}`)
   178→              }
   179→              lines.push("")
   180→              lines.push("**解析：**")
   181→              lines.push("")
   182→              lines.push(q.grading.explanation)
   183→              lines.push("")
   184→            }
   185→          })
   186→      }
   187→      lines.push("")
   188→    })
   189→    const blob = new Blob(["\uFEFF" + lines.join("\n")], { type: "text/markdown;charset=utf-8" })

   195→  const copyAll = async () => {
   196→    const text = targetImages
   197→      .flatMap((img, i) => {
   198→        const passages = img.questions.filter((q) => q.kind === "passage")

   298→  // ===== Word =====
   299→  const exportWord = async (targets: typeof images) => {
   300→    if (targets.length === 0) {
   301→      toast.error("没有图片")
   302→      return
   303→    }
   304→    setBusy(true)
   305→    try {
   306→      const docx = await import("docx")
   307→      const { Document, Packer, Paragraph, TextRun, ImageRun, HeadingLevel, AlignmentType } = docx
   308→      const children: any[] = []
   309→      for (let i = 0; i < targets.length; i++) {
   310→        const img = targets[i]
   311→        children.push(
   312→          new Paragraph({
   313→            text: `图片 ${i + 1}：${img.name}`,
   314→            heading: HeadingLevel.HEADING_1,
   315→          })
   316→        )
   317→        const passages = img.questions.filter((q) => q.kind === "passage")
   318→        passages.forEach((p) => {
   319→          children.push(
   320→            new Paragraph({
   321→              heading: HeadingLevel.HEADING_2,
   322→              children: [new TextRun({ text: "阅读材料", bold: true })],
   323→            })
   324→          )
   325→          p.content.split("\n").forEach((line) =>
   326→            children.push(new Paragraph({ text: line || " " }))
   327→          )
   328→        })
   329→        const questions = img.questions.filter((q) => q.kind !== "passage")
   330→        for (let qi = 0; qi < questions.length; qi++) {
   331→          const q = questions[qi]
   332→          const numText = q.number ? `第 ${q.number} 题` : `第 ${qi + 1} 题`
   333→          children.push(
   334→            new Paragraph({
   335→              heading: HeadingLevel.HEADING_2,
   336→              children: [new TextRun({ text: numText, bold: true })],
   337→            })
   338→          )
   339→          if (q.kind === "passage_question" && q.passageId) {
   340→            children.push(new Paragraph({ text: "（阅读材料见上）", italics: true }))
   341→          }
   342→          q.content.split("\n").forEach((line) =>
   343→            children.push(new Paragraph({ text: line || " " }))
   344→          )
   345→          // 题目截图（可选）
   346→          if (optIncludeImage) {
   347→          try {
   348→            const dataUrl = await cropToDataUrl(img, q.bbox, 800, true)
   349→            const base64 = dataUrl.split(",")[1]
   350→            children.push(
   351→              new Paragraph({
   352→                alignment: AlignmentType.CENTER,
   353→                children: [
   354→                  new ImageRun({
   355→                    data: Uint8Array.from(atob(base64), (c) => c.charCodeAt(0)),
   356→                    transformation: { width: 480, height: 180 },
   357→                  }),
   358→                ],
   359→              })
   360→            )
   361→          } catch (e) {
   362→            console.error(e)
   363→          }
   364→          }
   365→          // 配图（可选）
   366→          if (optIncludeSubImages) {
   367→          for (const si of q.images) {
   368→            try {
   369→              const subDataUrl = si.dataUrl ?? (await cropToDataUrl(img, si.bbox, 600, true))
   370→              if (subDataUrl) {
   371→                const base64 = subDataUrl.split(",")[1]
   372→                children.push(
   373→                  new Paragraph({
   374→                    alignment: AlignmentType.CENTER,
   375→                    children: [
   376→                      new ImageRun({
   377→                        data: Uint8Array.from(atob(base64), (c) => c.charCodeAt(0)),
   378→                        transformation: { width: 280, height: 200 },
   379→                      }),
   380→                    ],
   381→                  })
   382→                )
   383→                if (si.caption) {
   384→                  children.push(new Paragraph({
   385→                    alignment: AlignmentType.CENTER,
   386→                    children: [new TextRun({ text: si.caption, italics: true, size: 18 })],
   387→                  }))
   388→                }
   389→              }
   390→            } catch (e) {
   391→              console.error(e)
   392→            }
   393→          }
   394→          }
   395→          // 批改（可选）
   396→          if (optIncludeGrading && q.grading) {
   397→            const statusText =
   398→              q.grading.correct === true ? "✓ 正确" :
   399→              q.grading.correct === "partial" ? "△ 部分正确" : "✗ 错误"
   400→            children.push(
   401→              new Paragraph({
   402→                children: [
   403→                  new TextRun({
   404→                    text: `批改结果：${statusText}`,
   405→                    bold: true,
   406→                    color: q.grading.correct === true ? "16A34A" : q.grading.correct === "partial" ? "D97706" : "DC2626",
   407→                  }),
   408→                ],
   409→              })
   410→            )
   411→            if (q.grading.correctAnswer) {
   412→              children.push(new Paragraph({ text: `正确答案：${q.grading.correctAnswer}` }))
   413→            }
   414→            children.push(new Paragraph({ text: "解析：", bold: true }))
   415→            q.grading.explanation.split("\n").forEach((line) =>
   416→              children.push(new Paragraph({ text: line || " " }))
   417→            )
   418→          }
   419→        }
   420→      }
   421→      const doc = new Document({ sections: [{ children }] })
   422→      const blob = await Packer.toBlob(doc)
   423→      downloadBlob(blob, `questions-${Date.now()}.docx`)
   424→      toast.success(`已导出 ${targets.length} 张图的 Word 文档`)
   425→    } catch (e) {
   426→      toast.error(`Word 导出失败：${(e as Error).message}`)
   427→    } finally {
   428→      setBusy(false)
   429→    }
   430→  }

   432→  // ===== PDF：用 canvas 渲染中文，彻底解决乱码 =====
   433→  const exportPDF = async (targets: typeof images) => {
   434→    if (targets.length === 0) {
   435→      toast.error("没有图片")
   436→      return
   437→    }
   438→    setBusy(true)
   439→    try {
   440→      const { jsPDF } = await import("jspdf")
   441→      const pdf = new jsPDF({ unit: "pt", format: "a4", compress: true })
   442→      const pageW = pdf.internal.pageSize.getWidth()
   443→      const pageH = pdf.internal.pageSize.getHeight()
   444→      const margin = 24
   445→
   446→      for (let i = 0; i < targets.length; i++) {
   447→        const img = targets[i]
   448→        // 1. 第一页（可选）：原图 + 框 + 题号标签
   449→        if (optIncludeImage) {
   450→          const { renderImageWithBoxes } = await import("@/lib/merge-images")
   451→          const boxDataUrl = await renderImageWithBoxes(img, {
   452→            showLabels: true,
   453→            showGradingColor: optIncludeGrading,
   454→          })
   455→          const boxImg = await loadImageLocal(boxDataUrl)
   456→          const drawW = pageW - margin * 2
   457→          const drawH = (boxImg.naturalHeight / boxImg.naturalWidth) * drawW
   458→          const finalDrawH = Math.min(drawH, pageH - margin * 2)
   459→          const finalDrawW = (finalDrawH / drawH) * drawW
   460→          if (i > 0) pdf.addPage()
   461→          pdf.addImage(boxDataUrl, "JPEG", margin, margin, finalDrawW, finalDrawH)
   462→        }
   463→
   464→        // 2. 后续页：每道题渲染为图片（含文字 + 截图 + 配图 + 批改）
   465→        const questions = img.questions.filter((q) => q.kind !== "passage")
   466→        for (let qi = 0; qi < questions.length; qi++) {
   467→          const q = questions[qi]
   468→          // 找关联的阅读材料
   469→          let passageContent: string | undefined
   470→          if (q.kind === "passage_question" && q.passageId) {
   471→            const passage = img.questions.find((x) => x.id === q.passageId)
   472→            if (passage) passageContent = passage.content
   473→          }
   474→          try {
   475→            const { dataUrl: qDataUrl } = await renderQuestionToCanvas(img, q, passageContent, {
   476→              showGrading: optIncludeGrading,
   477→              showSubImages: optIncludeSubImages,
   478→              showQuestionImage: optIncludeImage,
   479→            })
   480→            if (qDataUrl) {
   481→              const qImg = await loadImageLocal(qDataUrl)
   482→              const qDrawW = pageW - margin * 2
   483→              const qDrawH = (qImg.naturalHeight / qImg.naturalWidth) * qDrawW
   484→              // 如果一页放不下，分块
   485→              if (qDrawH <= pageH - margin * 2) {
   486→                pdf.addPage()
   487→                pdf.addImage(qDataUrl, "JPEG", margin, margin, qDrawW, qDrawH)
   488→              } else {
   489→                // 分块画
   490→                const sliceCount = Math.ceil(qDrawH / (pageH - margin * 2))
   491→                const slicePxH = Math.floor(qImg.naturalHeight / sliceCount)
   492→                for (let s = 0; s < sliceCount; s++) {
   493→                  const sy = s * slicePxH
   494→                  const sh = s === sliceCount - 1 ? qImg.naturalHeight - sy : slicePxH
   495→                  const c2 = document.createElement("canvas")
   496→                  c2.width = qImg.naturalWidth
   497→                  c2.height = sh
   498→                  const c2ctx = c2.getContext("2d")
   499→                  if (!c2ctx) break
   500→                  c2ctx.drawImage(qImg, 0, sy, qImg.naturalWidth, sh, 0, 0, qImg.naturalWidth, sh)
   501→                  const sliceDataUrl = c2.toDataURL("image/jpeg", 0.85)
   502→                  const sliceDrawH = (sh / qImg.naturalWidth) * qDrawW
   503→                  pdf.addPage()
   504→                  pdf.addImage(sliceDataUrl, "JPEG", margin, margin, qDrawW, sliceDrawH)
   505→                }
   506→              }
   507→            }
   508→          } catch (e) {
   509→            console.error(`PDF 渲染题目 ${qi} 失败`, e)
   510→          }
   511→        }
   512→      }
   513→      pdf.save(`questions-${Date.now()}.pdf`)
   514→      toast.success(`已导出 ${targets.length} 张图的 PDF`)
   515→    } catch (e) {
   516→      toast.error(`PDF 导出失败：${(e as Error).message}`)
   517→    } finally {
   518→      setBusy(false)

   725→async function addLabelWatermark(
   726→  dataUrl: string,
   727→  label: string,
   728→  correct?: boolean | "partial"
   729→): Promise<string> {
   730→  const img = await loadImageLocal(dataUrl)
   731→  const canvas = document.createElement("canvas")
   732→  canvas.width = img.naturalWidth
   733→  canvas.height = img.naturalHeight
   734→  const ctx = canvas.getContext("2d")
   735→  if (!ctx) return dataUrl
   736→  ctx.drawImage(img, 0, 0)
   737→  if (label) {
   738→    const text = `第 ${label} 题`
   739→    ctx.font = "bold 24px sans-serif"
   740→    const tw = ctx.measureText(text).width
   741→    const bg =
   742→      correct === false ? "rgba(220, 38, 38, 0.95)" :
   743→      correct === "partial" ? "rgba(217, 119, 6, 0.95)" :
   744→      correct === true ? "rgba(22, 163, 74, 0.95)" :
   745→      "rgba(225, 29, 72, 0.95)"
   746→    ctx.fillStyle = bg
   747→    ctx.fillRect(8, 8, tw + 20, 32)
   748→    ctx.fillStyle = "white"
   749→    ctx.textBaseline = "middle"
   750→    ctx.fillText(text, 18, 24)
   751→  }
   752→  return canvas.toDataURL("image/png")
   753→}
   754→
   755→function loadImageLocal(dataUrl: string): Promise<HTMLImageElement> {
   756→  return new Promise((resolve, reject) => {
   757→    const im = new Image()
   758→    im.onload = () => resolve(im)
   759→    im.onerror = reject
   760→    im.src = dataUrl
   761→  })
   762→}
