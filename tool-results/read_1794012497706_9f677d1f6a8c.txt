   373→   373→                  new Paragraph({
   374→   374→                    alignment: AlignmentType.CENTER,
   375→   375→                    children: [
   376→   376→                      new ImageRun({
   377→   377→                        data: Uint8Array.from(atob(base64), (c) => c.charCodeAt(0)),

   379→   379→                      }),
   380→   380→                    ],
   381→   381→                  })

   400→   400→            children.push(
   401→   401→              new Paragraph({
   402→   402→                children: [
   403→   403→                  new TextRun({

   407→   407→                  }),
   408→   408→                ],
   409→   409→              })
   410→   410→            )
   411→   411→            if (q.grading.correctAnswer) {

   415→   415→            q.grading.explanation.split("\n").forEach((line) =>
   416→   416→              children.push(new Paragraph({ text: line || " " }))
   417→   417→            )
   418→   418→          }

   420→   420→      }
   421→   421→      const doc = new Document({ sections: [{ children }] })
   422→   422→      const blob = await Packer.toBlob(doc)
   423→   423→      downloadBlob(blob, `questions-${Date.now()}.docx`)
   424→   424→      toast.success(`已导出 ${targets.length} 张图的 Word 文档`)
   425→   425→    } catch (e) {
   426→   426→      toast.error(`Word 导出失败：${(e as Error).message}`)
   427→   427→    } finally {
   428→   428→      setBusy(false)
   429→   429→    }

   432→   432→  // ===== PDF：用 canvas 渲染中文，彻底解决乱码 =====
   433→   433→  const exportPDF = async (targets: typeof images) => {
   434→   434→    if (targets.length === 0) {
   435→   435→      toast.error("没有图片")
   436→   436→      return
   437→   437→    }

   442→   442→      const pageW = pdf.internal.pageSize.getWidth()
   443→   443→      const pageH = pdf.internal.pageSize.getHeight()

   725→   725→async function addLabelWatermark(
   726→   726→  dataUrl: string,
   727→   727→  label: string,
   728→   728→  correct?: boolean | "partial"
   729→   729→): Promise<string> {
   730→   730→  const img = await loadImageLocal(dataUrl)
   731→   731→  const canvas = document.createElement("canvas")
   732→   732→  canvas.width = img.naturalWidth
   733→   733→  canvas.height = img.naturalHeight
   734→   734→  const ctx = canvas.getContext("2d")
   735→   735→  if (!ctx) return dataUrl
   736→   736→  ctx.drawImage(img, 0, 0)
   737→   737→  if (label) {
   738→   738→    const text = `第 ${label} 题`
   739→   739→    ctx.font = "bold 24px sans-serif"
   740→   740→    const tw = ctx.measureText(text).width
   741→   741→    const bg =
   742→   742→      correct === false ? "rgba(220, 38, 38, 0.95)" :
   743→   743→      correct === "partial" ? "rgba(217, 119, 6, 0.95)" :
   744→   744→      correct === true ? "rgba(22, 163, 74, 0.95)" :
   745→   745→      "rgba(225, 29, 72, 0.95)"
   746→   746→    ctx.fillStyle = bg
   747→   747→    ctx.fillRect(8, 8, tw + 20, 32)
   748→   748→    ctx.fillStyle = "white"
   749→   749→    ctx.textBaseline = "middle"
   750→   750→    ctx.fillText(text, 18, 24)
   751→   751→  }
   752→   752→  return canvas.toDataURL("image/png")
   753→   753→}

   755→   755→function loadImageLocal(dataUrl: string): Promise<HTMLImageElement> {
   756→   756→  return new Promise((resolve, reject) => {
   757→   757→    const im = new Image()
   758→   758→    im.onload = () => resolve(im)
   759→   759→    im.onerror = reject
   760→   760→    im.src = dataUrl
   761→   761→  })
   762→   762→}
